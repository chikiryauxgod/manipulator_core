cmake_minimum_required(VERSION 3.20)
project(rtos_layer VERSION 0.1.0 LANGUAGES CXX)
add_library(rtos_layer INTERFACE)
add_library(rtos_layer::rtos_layer ALIAS rtos_layer)
target_include_directories(rtos_layer INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(rtos_layer INTERFACE cxx_std_17)
option(RTOS_LAYER_BUILD_TESTS "Build host API-contract tests using fake FreeRTOS" OFF)
if(RTOS_LAYER_BUILD_TESTS)
    enable_testing()
    add_executable(rtos_layer_tests tests/test.cpp)
    target_link_libraries(rtos_layer_tests PRIVATE rtos_layer)
    target_include_directories(rtos_layer_tests PRIVATE tests/fake_freertos)
    target_compile_options(rtos_layer_tests PRIVATE -Wall -Wextra -Werror -pedantic -fno-exceptions -fno-rtti)
    add_test(NAME rtos_layer_contract COMMAND rtos_layer_tests)
endif()

set(RTOS_LAYER_FREERTOS_PATH "" CACHE PATH "FreeRTOS-Kernel source for Linux integration test")
if(RTOS_LAYER_FREERTOS_PATH)
    enable_language(C)
    enable_testing()
    find_package(Threads REQUIRED)
    set(kernel ${RTOS_LAYER_FREERTOS_PATH})
    set(port ${kernel}/portable/ThirdParty/GCC/Posix)
    add_library(rtos_layer_test_kernel STATIC
        ${kernel}/tasks.c ${kernel}/queue.c ${kernel}/list.c
        ${kernel}/portable/MemMang/heap_4.c
        ${port}/port.c ${port}/utils/wait_for_event.c)
    target_include_directories(rtos_layer_test_kernel PUBLIC
        ${kernel}/include ${port} ${CMAKE_CURRENT_SOURCE_DIR}/tests)
    target_link_libraries(rtos_layer_test_kernel PUBLIC Threads::Threads)
    add_executable(rtos_layer_integration tests/integration.cpp)
    target_link_libraries(rtos_layer_integration PRIVATE rtos_layer rtos_layer_test_kernel)
    target_compile_options(rtos_layer_integration PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
    add_test(NAME rtos_layer_real_kernel COMMAND rtos_layer_integration)
    set_tests_properties(rtos_layer_real_kernel PROPERTIES TIMEOUT 10)
endif()
